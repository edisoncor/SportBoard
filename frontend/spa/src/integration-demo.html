Ejemplo de uso del servicio:

// Inyectar el servicio en un componente
constructor(private competenciesService: CompetenciesService) {}

// Obtener todos los catálogos
this.competenciesService.getCatalogues().subscribe({
  next: (catalogues) => {
    console.log('Catálogos obtenidos:', catalogues);
  },
  error: (error) => {
    console.error('Error:', error);
  }
});

// Crear un nuevo catálogo
const newCatalogue = {
  code: 'CAT001',
  description: 'Nuevo catálogo',
  parent_catalog: null
};

this.competenciesService.createCatalogue(newCatalogue).subscribe({
  next: (result) => {
    console.log('Catálogo creado:', result);
  }
});
